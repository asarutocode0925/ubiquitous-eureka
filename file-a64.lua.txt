local LEARN_MIN_SAMPLES = 10
local MAX_DEPTH = 20
local Learned
local function resetLearned()
    Learned = {dirKeys = {}, posKeys = {}, samples = 0, done = false}
end
resetLearned()

local function analyzePacket(packet)
    if Learned.done or not packet.args then return end
    Learned.samples = Learned.samples + 1
    if Learned.samples < LEARN_MIN_SAMPLES then return end
    local visited = {}
    local function walk(tbl, prefix, depth)
        if depth > MAX_DEPTH then return end
        if visited[tbl] then return end
        visited[tbl] = true
        for k, v in next, tbl do
            local key = prefix .. tostring(k)
            local tv = typeof(v)
            if tv == "Vector3" then
                local mg = v.Magnitude
                if mg >= 0.9 and mg <= 1.05 then
                    Learned.dirKeys[key] = true
                elseif mg > 5 then
                    if not Learned.dirKeys[key] then
                        Learned.posKeys[key] = true
                    end
                end
            elseif tv == "table" then
                walk(v, key .. ".", depth + 1)
            end
        end
    end
    walk(packet.args, "", 0)
    Learned.done = true
end

local function setByPath(tbl, path, value)
    local cur = tbl
    local last = nil
    for p in gmatch(path, "[^%.]+") do
        if last ~= nil then
            local k = tonumber(last) or last
            cur = cur[k]
            if type(cur) ~= "table" then return false end
        end
        last = p
    end
    if not last then return false end
    local k = tonumber(last) or last
    if cur[k] == nil then return false end
    cur[k] = value
    return true
end

local function rewritePacket(packet, camPos, targetPos)
    local delta = targetPos - camPos
    local mag = delta.Magnitude
    if mag < 0.01 then return 0 end
    local aimUnit = delta / mag
    local changed = 0
    for key in next, Learned.dirKeys do
        if setByPath(packet.args, key, aimUnit) then changed = changed + 1 end
    end
    for key in next, Learned.posKeys do
        if setByPath(packet.args, key, targetPos) then changed = changed + 1 end
    end
    return changed
end

    local id = plr:GetAttribute("TeamID")

local function pickPart(ch)
    local p = Cfg.part
    if p == "Head" then return ch:FindFirstChild("Head") end
    if p == "Torso" then
        return ch:FindFirstChild("UpperTorso") or ch:FindFirstChild("Torso") or ch:FindFirstChild("LowerTorso")
    end
    local cp = cam and cam.CFrame.Position or v3zero
    local b, bd = nil, huge
    for _, d in next, ch:GetDescendants() do
        if d:IsA("BasePart") and d.CanCollide then
            local dd = (d.Position - cp).Magnitude
            if dd < bd then bd = dd; b = d end
        end
    end
    return b
end

local function los(from, part, ch)
    if not Cfg.wall then return true end
    local d = part.Position - from
    local m = d.Magnitude
    if m < 0.01 then return true end
    wallParams.FilterDescendantsInstances = {LP.Character, ch}
    return O_Raycast(WSS, from, d, wallParams) == nil
end

local function nowPing()
    local ok, v = pcall(function()
        return Stats.Network.ServerStatsItem["Data Ping"]:GetValue() / 1000
    end)
    return ok and v or 0.05
end

local function isLockValid()
    local p = lockInfo.part
    if not p then return false end
    local parent = p.Parent
    if not parent then return false end
    local h = parent:FindFirstChildOfClass("Humanoid")
    return h and h.Health > 0
end

local function scan()
    local n = tick()
    if n - cache.t < 0.02 then return cache.v end
    cache.t = n
    if not cam or not cam.Parent then cam = WSS.CurrentCamera end
    if not cam or not isAlive(LP.Character) then cache.v = nil; return nil end

    local vp = cam.ViewportSize
    local ctrX, ctrY = vp.X * 0.5, vp.Y * 0.5
    local cp = cam.CFrame.Position
    local myTeam = getTeamId(LP)
    local fov2 = Cfg.fov * Cfg.fov
    local fovLimit2 = (Cfg.fov * 1.6) ^ 2
    local maxDist = Cfg.maxdist
    local wall = Cfg.wall
    local team = Cfg.team
    local partMode = Cfg.part

    if isLockValid() and (n - lockInfo.time) < Cfg.locktime then
        local ch = lockInfo.part.Parent
        local pos = lockInfo.part.Position
        local sp, on = cam:WorldToViewportPoint(pos)
        if sp.Z > 0 and on then
            local dx, dy = sp.X - ctrX, sp.Y - ctrY
            if dx * dx + dy * dy <= fovLimit2 then
                cache.v = {char = ch, part = lockInfo.part, pos = pos, plr = lockInfo.plr}
                return cache.v
            end
        end
        lockInfo.part = nil
    end

    local lead = Cfg.predict
    if Cfg.pingComp then lead = lead + nowPing() end

    local best, bestS = nil, huge
    for _, plr in next, Players:GetPlayers() do
        if plr ~= LP then
            local skip = false
            if team then
                local mt2 = myTeam
                if mt2 ~= nil then
                    local other = getTeamId(plr)
                    if other ~= nil and other == mt2 then skip = true end
                end
            end
            if not skip then
                local ch = plr.Character
                if isAlive(ch) then
                    local part
                    if partMode == "Head" then
                        part = ch:FindFirstChild("Head")
                    elseif partMode == "Torso" then
                        part = ch:FindFirstChild("UpperTorso") or ch:FindFirstChild("Torso") or ch:FindFirstChild("LowerTorso")
                    else
                        local b, bd = nil, huge
                        for _, d in next, ch:GetDescendants() do
                            if d:IsA("BasePart") and d.CanCollide then
                                local dd = (d.Position - cp).Magnitude
                                if dd < bd then bd = dd; b = d end
                            end
                        end
                        part = b
                    end
                    if part then
                        local pos = part.Position
                        local hrp = ch:FindFirstChild("HumanoidRootPart")
                        if hrp then
                            local vel = hrp.AssemblyLinearVelocity
                            if vel and vel.Magnitude >= 0.05 then
                                pos = pos + vel * lead
                            end
                        end
                        local dxw, dyw, dzw = pos.X - cp.X, pos.Y - cp.Y, pos.Z - cp.Z
                        local distSq = dxw * dxw + dyw * dyw + dzw * dzw
                        if distSq <= maxDist * maxDist then
                            local sp, on = cam:WorldToViewportPoint(pos)
                            if sp.Z > 0 and on then
                                local dx, dy = sp.X - ctrX, sp.Y - ctrY
                                local d2 = dx * dx + dy * dy
                                if d2 <= fov2 and d2 < bestS then
                                    if wall then
                                        wallParams.FilterDescendantsInstances = {LP.Character, ch}
                                        local rayD = pos - cp
                                        if O_Raycast(WSS, cp, rayD, wallParams) then
                                            goto continue
                                        end
                                    end
                                    bestS = d2
                                    best = {char = ch, part = part, pos = pos, plr = plr}
                                end
                            end
                        end
                    end
                end
            end
        end
        ::continue::
    end

    if best then
        lockInfo.part = best.part
        lockInfo.time = n
        lockInfo.plr = best.plr
    else
        lockInfo.part = nil
    end
    cache.v = best
    return best
end

local function getTarget()
    if scanBusy then return cache.v end
    scanBusy = true
    local ok, r = pcall(scan)
    scanBusy = false
    return ok and r or cache.v
end

local diagLog = {}
local MAX_LOG = 20
local logDirty = false
local function pushLog(s)
    insert(diagLog, s)
    if #diagLog > MAX_LOG then remove(diagLog, 1) end
    logDirty = true
end

-- メタテーブルフックを安全に実行
local mtHookSuccess = false
if getrawmetatable and setreadonly and (newcclosure or true) then
    local ok, mt = pcall(getrawmetatable, game)
    if ok and mt then
        local oldNC = mt.__namecall
        local function newNamecall(self, ...)
            if _checkcaller() then return oldNC(self, ...) end
            if self == UseItem and getnamecallmethod() == "FireServer" then
                local args = {...}
                local packet, pIdx
                for i = 1, #args do
                    local a = args[i]
                    if type(a) == "table" and a.type ~= nil then
                        packet = a; pIdx = i; break
                    end
                end
                if packet and packet.type == "StartShooting" then
                    if not Learned.done then
                        if Cfg.autoLearn then analyzePacket(packet) end
                        return oldNC(self, ...)
                    end
                    if Cfg.on and packet.args then
                        local t = getTarget()
                        if t then
                            local cp = cam and cam.CFrame.Position or v3zero
                            local dx, dy, dz = t.pos.X - cp.X, t.pos.Y - cp.Y, t.pos.Z - cp.Z
                            if dx * dx + dy * dy + dz * dz > 1 then
                                local n = rewritePacket(packet, cp, t.pos)
                                pushLog(format("[REWRITE] %d -> %s", n, t.plr.Name))
                                if n > 0 then
                                    args[pIdx] = packet
                                    return oldNC(self, unpack(args, 1, #args))
                                end
                            end
                        end
                    end
                end
            end
            return oldNC(self, ...)
        end
        
        pcall(function()
            setreadonly(mt, false)
            mt.__namecall = newcclosure and newcclosure(newNamecall) or newNamecall
            setreadonly(mt, true)
            mtHookSuccess = true
        end)
    end
end

-- ドラッグ機能（マウス＆タッチ対応）
local dragState = {active = false, target = nil, start = nil, startPos = nil}

local function beginDrag(frame, input)
    dragState.active = true
    dragState.target = frame
    dragState.start = input.Position
    dragState.startPos = frame.Position
end

header.InputBegan:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.Touch or input.UserInputType == Enum.UserInputType.MouseButton1 then
        beginDrag(main, input)
    end
end)

UIS.InputChanged:Connect(function(input)
    if dragState.active and (input.UserInputType == Enum.UserInputType.Touch or input.UserInputType == Enum.UserInputType.MouseMovement) then
        local d = input.Position - dragState.start
        local sp = dragState.startPos
        dragState.target.Position = UDim2.new(sp.X.Scale, sp.X.Offset + d.X, sp.Y.Scale, sp.Y.Offset + d.Y)
    end
end)

UIS.InputEnded:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.Touch or input.UserInputType == Enum.UserInputType.MouseButton1 then
        dragState.active = false
        dragState.target = nil
    end
end)

local activeSlider = nil

local function makeSlider(parent, label, minV, maxV, get, set, fmt, order)
    local row = makeCard(parent, 54)
    row.LayoutOrder = order

    new("TextLabel", {
        Size = UDim2.new(0.6, -14, 0, 20),
        Position = UDim2.new(0, 14, 0, 5),
        BackgroundTransparency = 1,
        Text = label,
        TextColor3 = Theme.text,
        Font = Enum.Font.GothamMedium,
        TextSize = 13,
        TextXAlignment = Enum.TextXAlignment.Left,
    }, row)

    local valLbl = new("TextLabel", {
        Size = UDim2.new(0.4, -14, 0, 20),
        Position = UDim2.new(0.6, 0, 0, 5),
        BackgroundTransparency = 1,
        Text = fmt and fmt(get()) or tostring(get()),
        TextColor3 = Theme.accent,
        Font = Enum.Font.GothamBold,
        TextSize = 13,
        TextXAlignment = Enum.TextXAlignment.Right,
    }, row)

    local track = new("Frame", {
        Size = UDim2.new(1, -28, 0, 6),
        Position = UDim2.new(0, 14, 0, 34),
        BackgroundColor3 = Theme.stroke,
        BorderSizePixel = 0,
    }, row)
    corner(track, UDim.new(1, 0))

    local fill = new("Frame", {
        Size = UDim2.new((get() - minV) / (maxV - minV), 0, 1, 0),
        BackgroundColor3 = Theme.accent,
        BorderSizePixel = 0,
        ZIndex = 2,
    }, track)
    corner(fill, UDim.new(1, 0))
    addGrad(fill, Theme.accent, Theme.accent2, 0)

    local knob = new("Frame", {
        Size = UDim2.fromOffset(16, 16),
        Position = UDim2.new((get() - minV) / (maxV - minV), 0, 0.5, 0),
        AnchorPoint = v2(0.5, 0.5),
        BackgroundColor3 = Color3.fromRGB(255, 255, 255),
        BorderSizePixel = 0,
        ZIndex = 3,
    }, track)
    corner(knob, UDim.new(1, 0))

    local hit = new("TextButton", {
        Size = UDim2.new(1, 0, 1, 0),
        BackgroundTransparency = 1,
        Text = "",
        ZIndex = 4,
    }, row)

    local function updateSlider(input)
        local tw_ = track.AbsoluteSize.X
        if tw_ > 0 then
            local rel = clamp((input.Position.X - track.AbsolutePosition.X) / tw_, 0, 1)
            local val = minV + (maxV - minV) * rel
            if maxV - minV >= 5 then val = floor(val + 0.5) end
            set(val)
            fill.Size = UDim2.new(rel, 0, 1, 0)
            knob.Position = UDim2.new(rel, 0, 0.5, 0)
            valLbl.Text = fmt and fmt(val) or tostring(val)
        end
    end

    hit.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.Touch or input.UserInputType == Enum.UserInputType.MouseButton1 then
            activeSlider = {track = track, update = updateSlider}
            updateSlider(input)
        end
    end)

    return row
end

UIS.InputChanged:Connect(function(input)
    if activeSlider and (input.UserInputType == Enum.UserInputType.Touch or input.UserInputType == Enum.UserInputType.MouseMovement) then
        activeSlider.update(input)
    end
end)

UIS.InputEnded:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.Touch or input.UserInputType == Enum.UserInputType.MouseButton1 then
        activeSlider = nil
    end
end)

local logLabelPool = {}

local function refreshLog()
    local need = #diagLog
    local have = #logLabelPool

    if have < need then
        for i = have + 1, need do
            local lbl = new("TextLabel", {
                Size = UDim2.new(1, -4, 0, 15),
                BackgroundTransparency = 1,
                TextColor3 = Theme.text,
                Font = Enum.Font.Code,
                TextSize = 11,
                TextXAlignment = Enum.TextXAlignment.Left,
            }, logScroll)
            lbl.LayoutOrder = i
            logLabelPool[i] = lbl
        end
    elseif have > need then
        for i = have, need + 1, -1 do
            pcall(function() logLabelPool[i]:Destroy() end)
            logLabelPool[i] = nil
        end
    end

    for i = 1, need do
        logLabelPool[i].Text = diagLog[i]
        logLabelPool[i].LayoutOrder = i
    end

    logScroll.CanvasSize = UDim2.new(0, 0, 0, need * 17 + 8)
    if Cfg.logOpen then
        logScroll.CanvasPosition = v2(0, logScroll.AbsoluteCanvasSize.Y)
    end
end

local wasDone = false
local logAccum = 0
local uiAccum = 0

RunService.Heartbeat:Connect(function(dt)
    if logDirty then
        logAccum = logAccum + dt
        if logAccum >= 0.1 then
            logAccum = 0
            logDirty = false
            refreshLog()
        end
    end

    if Learned.done and not wasDone then
        wasDone = true
        local dn, pn = 0, 0
        for _ in next, Learned.dirKeys do dn = dn + 1 end
        for _ in next, Learned.posKeys do pn = pn + 1 end
        learnLbl.Text = format("Learned · %d dir / %d pos", dn, pn)
        activateToggle()
    elseif not Learned.done then
        learnLbl.Text = format("Learning %d / %d", Learned.samples, LEARN_MIN_SAMPLES)
    end

    if not cam then return end

    if Cfg.showFov then
        local vp = cam.ViewportSize
        fovCircle.Position = UDim2.fromOffset(vp.X * 0.5, vp.Y * 0.5)
    end

    if Cfg.showBox then
        uiAccum = uiAccum + dt
        if uiAccum >= 0.033 then
            uiAccum = 0
            local t = getTarget()
            if t and t.part and t.part.Parent then
                local sp, on = cam:WorldToViewportPoint(t.part.Position)
                if sp.Z > 0 and on then
                    targetBox.Position = UDim2.fromOffset(sp.X - 32, sp.Y - 48)
                    targetBox.Size = UDim2.fromOffset(64, 96)
                    targetBox.Visible = true
                elseif targetBox.Visible then
                    targetBox.Visible = false
                end
            elseif targetBox.Visible then
                targetBox.Visible = false
            end
        end
    elseif targetBox.Visible then
        targetBox.Visible = false
    end
end)
